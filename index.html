const express = require("express"); const bcrypt = require("bcrypt"); const jwt = require("jsonwebtoken"); const app = express(); app.use(express.json()); const user = { username: "demoUser", passwordHash: bcrypt.hashSync("StrongPassword123!", 10) }; app.post("/login", async (req, res) => { const { username, password } = req.body; if (username !== user.username) { return res.status(401).json({ message: "Invalid credentials" }); } const valid = await bcrypt.compare(password, user.passwordHash); if (!valid) { return res.status(401).json({ message: "Invalid credentials" }); } const token = jwt.sign( { username }, process.env.JWT_SECRET, { expiresIn: "1h" } ); res.json({ message: "Login successful", token }); }); app.listen(3000, () => { console.log("Server running on port 3000"); });
